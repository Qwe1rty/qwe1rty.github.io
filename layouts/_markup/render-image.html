"" | replaceRE "var\\(--ge-adaptive-fg,\\s*#000000\\)" "#000000" | resources.FromString (path.Join (path.Dir $img.RelPermalink) (path.Base $img.Name)) }} {{- else }} {{- $srcset := slice }} {{- $default := false }} {{- range slice 480 768 1024 }} {{- if le . $img.Width }} {{- $r := $img.Resize (printf "%dx webp q80" .) }} {{- $srcset = $srcset | append (printf "%s %dw" $r.RelPermalink .) }} {{- $default = $r }} {{- end }} {{- end }} {{- /* Always offer the original as the largest srcset entry / fallback src. */}} {{- $srcset = $srcset | append (printf "%s %dw" $img.RelPermalink $img.Width) }} {{- if not $default }}{{ $default = $img }}{{ end }} {{- end }} {{- else }} {{- end }}
